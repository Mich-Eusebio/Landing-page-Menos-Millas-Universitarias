        function openModal(planKey) {
            const modal = document.getElementById('modal-beneficios');
            const menu = document.getElementById('tab-menu-container');
            const beneficios = getBeneficiosPlan(planKey);
            
            document.getElementById('content-subtitle').innerText = nombresPlanes[planKey];
            menu.innerHTML = '';
            
            beneficios.forEach((tab, index) => {
                const btn = document.createElement('button');
                btn.className = `tab-link ${index === 0 ? 'active' : ''}`;
                btn.innerText = tab.label;
                btn.onclick = () => {
                    document.querySelectorAll('.tab-link').forEach(l => l.classList.remove('active'));
                    btn.classList.add('active');
                    updateContent(tab);
                };
                menu.appendChild(btn);
            });
            updateContent(beneficios[0]);
            modal.style.display = 'flex';
            document.body.style.overflow = 'hidden';
        }

        function updateContent(tab) {
            document.getElementById('content-title').innerText = tab.title;
            document.getElementById('content-body').innerHTML = tab.content;
        }

        function closeModal() {
            document.getElementById('modal-beneficios').style.display = 'none';
            document.body.style.overflow = 'auto';
        }

        window.onclick = function(event) { if (event.target == document.getElementById('modal-beneficios')) closeModal(); }
